# Agent-KB P0 骨架 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 搭建 Agent-KB 知识仓骨架并交付 `kb` CLI（new/lint/index/search/show/related/stat/promote），实现零外部服务的"规范 + 模板 + 校验 + 索引 + 检索"闭环。

**Architecture:** git + Markdown 为唯一事实源；`taxonomy.yaml` 是分类法唯一事实源并派生 JSON Schema；每条知识是一个带 YAML front-matter 的原子 `.md`；CLI 基于 front-matter 元数据生成 `INDEX.md`（人读）与 `index.jsonl`（机读），检索只返回元数据+摘要以省 token。通用知识在 Agent-KB，项目知识在各项目 `docs/kb/`，二者共用同一套 schema/模板/CLI。

**Tech Stack:** Python 3.10+、`pyyaml`（front-matter 解析）、`jsonschema`（校验）、`pytest`（测试）。无 Web/DB/向量依赖。

## Global Constraints

以下为全局约束，每个任务的实现都隐式包含本节：

- Python 版本下限 3.10；所有源文件顶部不写多余 shebang（可执行入口除外）。
- 第三方依赖仅限 `pyyaml`、`jsonschema`（运行时）与 `pytest`（开发）。P0 严禁引入向量库、MCP、SaaS SDK。
- 日期字段一律 `YYYY-MM-DD`；展示给终端的时间用 `YYYY-MM-DD HH:mm:ss`。
- `id` 格式固定 `kb-<yyyyMMdd>-<4hex>`，正则 `^kb-\d{8}-[0-9a-f]{4}$`；生成后不可变。
- 正文路径规则 `kb/<type>/<domain>/<slug>.md`，目录段**直接取 `type`/`domain` 枚举原值**（单数、不映射复数）；`<slug>` = `YYYY-MM-DD-<kebab-case>`。
- `type` ∈ {decision, spec, plan, pitfall, rca, howto, reference, retrospective}；`domain` ∈ {backend, frontend, database, devops, security, testing, build, algorithm, process, tooling}；`scope` ∈ {project, stack, universal}；`status` ∈ {draft, proposed, accepted, deprecated, superseded}；`severity` ∈ {low, medium, high, critical}；`confidence` ∈ {verified, probable, speculative}。枚举以 `taxonomy.yaml` 为准。
- CLI 遵循 spec §11.2 命令语义；`search` 默认 `--limit 20`、只输出元数据+摘要+路径，绝不输出正文。
- `status: draft` 条目不进 `index.jsonl`，写入 `index.draft.jsonl`，检索默认排除。
- 遵循工作区 CLAUDE.md：严禁 DDL；本计划不含任何数据库操作。
- 提交信息用英文、小写命令式（`feat:` / `test:` / `chore:` / `docs:`）。

```python
from __future__ import annotations
import functools
import pathlib
import yaml

_ID_PATTERN = r"^kb-\d{8}-[0-9a-f]{4}$"
_DATE_PATTERN = r"^\d{4}-\d{2}-\d{2}$"


def repo_root() -> pathlib.Path:
    here = pathlib.Path(__file__).resolve()
    for parent in here.parents:
        if (parent / "taxonomy.yaml").exists():
            return parent
    return here.parents[2]


@functools.lru_cache(maxsize=1)
def _default_path() -> pathlib.Path:
    return repo_root() / "taxonomy.yaml"


def load_taxonomy(path: pathlib.Path | None = None) -> dict:
    p = pathlib.Path(path) if path else _default_path()
    with open(p, encoding="utf-8") as f:
        return yaml.safe_load(f)


def derive_schema(tax: dict) -> dict:
    types = sorted(tax["types"])
    domains = sorted(tax["domains"])
    enums = tax["enums"]
    props = {
        "id": {"type": "string", "pattern": _ID_PATTERN},
        "title": {"type": "string", "minLength": 1},
        "type": {"type": "string", "enum": types},
        "domain": {
            "type": "array", "minItems": 1,
            "items": {"type": "string", "enum": domains},
        },
        "tags": {"type": "array", "items": {"type": "string"}},
        "scope": {"type": "string", "enum": enums["scope"]},
        "status": {"type": "string", "enum": enums["status"]},
        "severity": {"type": "string", "enum": enums["severity"]},
        "confidence": {"type": "string", "enum": enums["confidence"]},
        "created": {"type": "string", "pattern": _DATE_PATTERN},
        "updated": {"type": "string", "pattern": _DATE_PATTERN},
        "review_by": {"type": ["string", "null"], "pattern": _DATE_PATTERN},
        "source": {
            "type": "object",
            "properties": {"project": {"type": "string"}, "ref": {"type": "string"}},
        },
        "related": {"type": "array", "items": {"type": "string"}},
        "supersedes": {"type": ["string", "null"]},
        "superseded_by": {"type": ["string", "null"]},
        "dedup_key": {"type": ["string", "null"]},
    }
    return {
        "$schema": "https://json-schema.org/draft/2020-12/schema",
        "title": "Agent-KB front-matter",
        "type": "object",
        "required": ["id", "title", "type", "domain", "scope",
                      "status", "confidence", "created", "updated"],
        "properties": props,
    }
```

```python
from __future__ import annotations
import datetime as _dt
import pathlib
import re
from dataclasses import dataclass
import yaml

_FM_RE = re.compile(r"^---\s*\n(.*?)\n---\s*\n?(.*)$", re.DOTALL)


def _normalize_dates(fm: dict) -> dict:
    # YAML 会把未加引号的 2026-10-12 解析成 datetime.date; schema 要求 str, 统一归一为 ISO 字符串
    for k, v in fm.items():
        if isinstance(v, _dt.datetime):
            fm[k] = v.date().isoformat()
        elif isinstance(v, _dt.date):
            fm[k] = v.isoformat()
    return fm


@dataclass
class Entry:
    fm: dict
    body: str
    path: pathlib.Path

    def h1(self) -> str | None:
        for line in self.body.splitlines():
            s = line.strip()
            if s.startswith("# "):
                return s[2:].strip()
        return None

    def summary(self, limit: int = 120) -> str:
        for block in re.split(r"\n\s*\n", self.body):
            b = block.strip()
            if not b or b.startswith("#"):
                continue
            b = re.sub(r"\s+", " ", b)
            return b[:limit]
        return ""


def parse_entry(path: pathlib.Path) -> Entry:
    text = pathlib.Path(path).read_text(encoding="utf-8")
    m = _FM_RE.match(text)
    if not m:
        return Entry(fm={}, body=text, path=pathlib.Path(path))
    fm = _normalize_dates(yaml.safe_load(m.group(1)) or {})
    body = m.group(2)
    return Entry(fm=fm, body=body, path=pathlib.Path(path))


def dump_entry(entry: Entry) -> str:
    fm_text = yaml.safe_dump(entry.fm, allow_unicode=True, sort_keys=False).strip()
    body = entry.body if entry.body.startswith("\n") else "\n" + entry.body
    return f"---\n{fm_text}\n---\n{body}"
```

```python
    return f"kb-{today.strftime('%Y%m%d')}-{secrets.token_hex(2)}"


def slug(title: str, today: dt.date) -> str:
    k = _NONKEBAB.sub("-", title.lower()).strip("-")
    return f"{today.isoformat()}-{k}" if k else today.isoformat()


def _template_body(type_name: str) -> str:
    tax = load_taxonomy()
    sections = tax["types"][type_name]["sections"]
    return "\n".join(f"## {s}\n<!-- TODO -->\n" for s in sections)


def create_entry(repo_root: Path, type: str, domain: str, scope: str,
                 title: str, generated: bool, today: dt.date) -> Entry:
    tax = load_taxonomy()
    if type not in tax["types"]:
        raise ValueError(f"unknown type: {type}")
    if domain not in tax["domains"]:
        raise ValueError(f"unknown domain: {domain}")
    fm = {
        "id": new_id(today),
        "title": title,
        "type": type,
        "domain": [domain],
        "tags": [],
        "scope": scope,
        "status": "draft" if generated else "proposed",
        "confidence": "speculative" if generated else "probable",
        "created": today.isoformat(),
        "updated": today.isoformat(),
    }
    if "severity" in tax["types"][type].get("requires", []):
        fm["severity"] = "medium"
    fm_text = "\n".join(f"{k}: {v}" for k, v in fm.items())
    body = f"\n# {title}\n\n{_template_body(type)}"
    rel = Path("kb") / type / domain / f"{slug(title, today)}.md"
    path = Path(repo_root) / rel
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_text(f"---\n{fm_text}\n---\n{body}", encoding="utf-8")
    return parse_entry(path)
```

```python
from __future__ import annotations
from pathlib import Path
import jsonschema
from kb.entry import parse_entry
from kb.taxonomy import load_taxonomy, derive_schema


def _iter_entries(root: Path):
    kb = Path(root) / "kb"
    if not kb.exists():
        return
    for p in sorted(kb.rglob("*.md")):
        yield parse_entry(p)


def lint_repo(repo_root: Path, scope: str = "project") -> list[str]:
    tax = load_taxonomy()
    schema = derive_schema(tax)
    validator = jsonschema.Draft202012Validator(schema)
    errors: list[str] = []
    entries = list(_iter_entries(repo_root))
    ids = {e.fm.get("id") for e in entries}
    for e in entries:
        rid = e.fm.get("id", e.path.name)
        for err in sorted(validator.iter_errors(e.fm), key=lambda x: x.json_path):
            errors.append(f"{rid}: schema {err.json_path}: {err.message}")
        if e.fm.get("scope") == "project" and scope == "agent-kb":
            errors.append(f"{rid}: scope=project must not live in agent-kb")
        if e.fm.get("scope") != "project" and scope == "project":
            errors.append(f"{rid}: scope!=project must be promoted to agent-kb")
        for f in ("related",):
            for ref in e.fm.get(f, []) or []:
                if ref not in ids:
                    errors.append(f"{rid}: {f} references missing {ref}")
        for f in ("supersedes", "superseded_by"):
            ref = e.fm.get(f)
            if ref and ref not in ids:
                errors.append(f"{rid}: {f} references missing {ref}")
        h1 = sum(1 for ln in e.body.splitlines() if ln.strip().startswith("# "))
        if h1 != 1:
            errors.append(f"{rid}: expected exactly one H1, found {h1}")
    return errors
```

```python
from __future__ import annotations
import hashlib
import json
from pathlib import Path
from kb.entry import parse_entry


def _records(repo_root: Path, repo_name: str):
    kb = Path(repo_root) / "kb"
    live, draft = [], []
    if not kb.exists():
        return live, draft
    for p in sorted(kb.rglob("*.md")):
        e = parse_entry(p)
        fm = e.fm
        if not fm.get("id"):
            continue
        rec = {
            "id": fm["id"], "title": fm.get("title", ""),
            "type": fm.get("type", ""), "domain": fm.get("domain", []),
            "tags": fm.get("tags", []), "scope": fm.get("scope", ""),
            "status": fm.get("status", ""), "severity": fm.get("severity"),
            "confidence": fm.get("confidence", ""),
            "path": p.relative_to(repo_root).as_posix(),
            "summary": e.summary(),
            "dedup_key": fm.get("dedup_key"),
            "related": fm.get("related", []),
            "updated": fm.get("updated", ""), "repo": repo_name,
        }
        (draft if fm.get("status") == "draft" else live).append(rec)
    return live, draft


def _jsonl(records) -> str:
    return "".join(json.dumps(r, ensure_ascii=False) + "\n" for r in records)


def _md(records) -> str:
    lines = ["# Knowledge Index", "",
             "> Generated by `kb index`. Do not edit by hand.", ""]
    for r in records:
        lines.append(
            f"- [{r['title']}]({r['path']}) · {r['type']} · "
            f"{','.join(r['domain'])} · {r['status']} · {r['summary']}"
        )
    return "\n".join(lines) + "\n"


def build_indexes(repo_root: Path, repo_name: str = "agent-kb"):
    live, draft = _records(repo_root, repo_name)
    root = Path(repo_root) / "kb"
    root.mkdir(parents=True, exist_ok=True)
    md = root / "INDEX.md"
    jl = root / "index.jsonl"
    dj = root / "index.draft.jsonl"
    md.write_text(_md(live), encoding="utf-8")
    jl.write_text(_jsonl(live), encoding="utf-8")
    dj.write_text(_jsonl(draft), encoding="utf-8")
    return md, jl, dj


def check_indexes(repo_root: Path, repo_name: str = "agent-kb") -> bool:
    live, draft = _records(repo_root, repo_name)
    root = Path(repo_root) / "kb"
    want = {root / "INDEX.md": _md(live), root / "index.jsonl": _jsonl(live),
            root / "index.draft.jsonl": _jsonl(draft)}
    for path, content in want.items():
        if not path.exists() or path.read_text(encoding="utf-8") != content:
            return False
    return True
```

```python
from __future__ import annotations
import json
from pathlib import Path


def _load(repo_root: Path):
    jl = Path(repo_root) / "kb" / "index.jsonl"
    if not jl.exists():
        return []
    return [json.loads(x) for x in jl.read_text(encoding="utf-8").splitlines() if x.strip()]


def search(repo_root: Path, type=None, domain=None, tag=None, scope=None,
           status=None, confidence=None, query=None, limit: int = 20) -> list[dict]:
    recs = _load(repo_root)
    out = []
    q = query.lower() if query else None
    for r in recs:
        if type and r["type"] != type:
            continue
        if domain and domain not in r["domain"]:
            continue
        if tag and tag not in r.get("tags", []):
            continue
        if scope and r["scope"] != scope:
            continue
        if status and r["status"] != status:
            continue
        if confidence and r["confidence"] != confidence:
            continue
        if q:
            hay = f"{r['title']} {' '.join(r.get('tags', []))} {r['summary']}".lower()
            if q not in hay:
                continue
        out.append(r)
    return out[:limit]
```

```python
from __future__ import annotations
import json
import datetime as dt
from collections import Counter
from pathlib import Path
from kb.entry import parse_entry


def _records(repo_root: Path):
    jl = Path(repo_root) / "kb" / "index.jsonl"
    dj = Path(repo_root) / "kb" / "index.draft.jsonl"
    recs = []
    for f in (jl, dj):
        if f.exists():
            recs += [json.loads(x) for x in f.read_text(encoding="utf-8").splitlines() if x.strip()]
    return recs


def _find_path(repo_root: Path, id: str) -> Path | None:
    for r in _records(repo_root):
        if r["id"] == id:
            return Path(repo_root) / r["path"]
    return None


def get_entry(repo_root: Path, id: str) -> str | None:
    p = _find_path(repo_root, id)
    return p.read_text(encoding="utf-8") if p else None


def related(repo_root: Path, id: str) -> list[dict]:
    recs = _records(repo_root)
    by_id = {r["id"]: r for r in recs}
    self = by_id.get(id)
    out, seen = [], set()
    if self:
        for rid in self.get("related", []):
            if rid in by_id and rid not in seen:
                out.append(by_id[rid]); seen.add(rid)
    for r in recs:
        if id in r.get("related", []) and r["id"] not in seen:
            out.append(r); seen.add(r["id"])
    return out


def stat(repo_root: Path) -> dict:
    recs = _records(repo_root)
    due = due_review(repo_root, dt.date.today())
    return {
        "total": len(recs),
        "by_type": dict(Counter(r["type"] for r in recs)),
        "by_domain": dict(Counter(d for r in recs for d in r["domain"])),
        "by_status": dict(Counter(r["status"] for r in recs)),
        "by_confidence": dict(Counter(r["confidence"] for r in recs)),
        "draft_backlog": sum(1 for r in recs if r["status"] == "draft"),
        "due_review": len(due),
    }


def due_review(repo_root: Path, today: dt.date) -> list[dict]:
    out = []
    for p in sorted((Path(repo_root) / "kb").rglob("*.md")):
        e = parse_entry(p)
        rb = e.fm.get("review_by")
        if rb and str(rb) <= today.isoformat():
            out.append({"id": e.fm.get("id"), "title": e.fm.get("title"),
                        "review_by": str(rb)})
    return out
```

```python
            return p
    return None


def promote(repo_root: Path, id: str, to: str) -> dict:
    path = _find(repo_root, id)
    if not path:
        raise FileNotFoundError(f"entry not found: {id}")
    e = parse_entry(path)
    e.fm["scope"] = to
    tags = list(e.fm.get("tags", []) or [])
    if "promoted" not in tags:
        tags.append("promoted")
    e.fm["tags"] = tags
    path.write_text(dump_entry(e), encoding="utf-8")
    return e.fm
```

```python
from __future__ import annotations
import argparse
import datetime as dt
import json
import sys
from pathlib import Path
from kb.new_entry import create_entry
from kb.lint import lint_repo
from kb.index import build_indexes, check_indexes
from kb.search import search
from kb.query import get_entry, related, stat, due_review
from kb.promote import promote


def main(argv: list[str] | None = None) -> int:
    ap = argparse.ArgumentParser(prog="kb")
    common = argparse.ArgumentParser(add_help=False)
    common.add_argument("--repo", default=".", help="repo root containing kb/")
    sub = ap.add_subparsers(dest="cmd", required=True)

    p_new = sub.add_parser("new", parents=[common])
    p_new.add_argument("type")
    p_new.add_argument("--domain", required=True)
    p_new.add_argument("--scope", required=True)
    p_new.add_argument("--title", required=True)
    p_new.add_argument("--generated", action="store_true")

    sub.add_parser("lint", parents=[common]).add_argument("--scope", default="project")
    p_idx = sub.add_parser("index", parents=[common])
    p_idx.add_argument("--check", action="store_true")

    p_s = sub.add_parser("search", parents=[common])
    for flag in ("type", "domain", "tag", "scope", "status", "confidence", "query"):
        p_s.add_argument(f"--{flag}", default=None)
    p_s.add_argument("--limit", type=int, default=20)

    p_show = sub.add_parser("show", parents=[common]); p_show.add_argument("id")
    p_rel = sub.add_parser("related", parents=[common]); p_rel.add_argument("id")
    p_stat = sub.add_parser("stat", parents=[common]); p_stat.add_argument("--due-review", action="store_true")
    p_pro = sub.add_parser("promote", parents=[common]); p_pro.add_argument("id"); p_pro.add_argument("--to", required=True)

    args = ap.parse_args(argv)
    repo = Path(args.repo)

    if args.cmd == "new":
        e = create_entry(repo, type=args.type, domain=args.domain, scope=args.scope,
                         title=args.title, generated=args.generated, today=dt.date.today())
        print(e.path)
        return 0
    if args.cmd == "lint":
        errs = lint_repo(repo, scope=args.scope)
        for e in errs:
            print(e, file=sys.stderr)
        return 1 if errs else 0
    if args.cmd == "index":
        if args.check:
            ok = check_indexes(repo)
            print("index up to date" if ok else "index stale")
            return 0 if ok else 1
        md, jl, dj = build_indexes(repo)
        print(f"wrote {md}\nwrote {jl}\nwrote {dj}")
        return 0
    if args.cmd == "search":
        res = search(repo, type=args.type, domain=args.domain, tag=args.tag,
                     scope=args.scope, status=args.status, confidence=args.confidence,
                     query=args.query, limit=args.limit)
        print(json.dumps(res, ensure_ascii=False, indent=2))
        return 0
    if args.cmd == "show":
        text = get_entry(repo, args.id)
        print(text if text else f"not found: {args.id}")
        return 0 if text else 1
    if args.cmd == "related":
        print(json.dumps(related(repo, args.id), ensure_ascii=False, indent=2))
        return 0
    if args.cmd == "stat":
        if args.due_review:
            print(json.dumps(due_review(repo, dt.date.today()), ensure_ascii=False, indent=2))
        else:
            print(json.dumps(stat(repo), ensure_ascii=False, indent=2))
        return 0
    if args.cmd == "promote":
        fm = promote(repo, args.id, args.to)
        print(json.dumps(fm, ensure_ascii=False, indent=2))
        return 0
    return 2


if __name__ == "__main__":
    raise SystemExit(main())
```
