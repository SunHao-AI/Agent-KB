```python
from __future__ import annotations
import hashlib
import re
from kb.entry import Entry

# 规范化删除的标点集(中英文, 固定不可配置, spec §4)
_PUNCT = (
    ",.;:!?\"'`()[]{}<>`~@#$%^&*-_=+/\\|"
    "，。、；：！？“”‘’（）《》【】〔〕「」『』…—～·〈〉"
)
_STRIP_RE = re.compile(f"[{re.escape(_PUNCT)}\\s]")


def normalize(text: str) -> str:
    return _STRIP_RE.sub("", text.lower())


def fingerprint_source(entry: Entry) -> str:
    """第一个 H2 含'根因'的段全文; 无根因段则退化正文全文(剔 H1 行)."""
    out: list[str] = []
    capturing = False
    in_fence = False
    lines = entry.body.splitlines()
    for ln in lines:
        if ln.strip().startswith("```"):
            in_fence = not in_fence
            if capturing:
                out.append(ln)
            continue
        if not in_fence and ln.startswith("## "):
            if capturing:
                break
            if "根因" in ln:
                capturing = True
            continue
        if capturing:
            out.append(ln)
    if capturing:
        return "\n".join(out).strip()
    return "\n".join(ln for ln in lines if not ln.strip().startswith("# ")).strip()


def fingerprint(entry: Entry) -> str:
    src = normalize(fingerprint_source(entry))
    return hashlib.sha256(src.encode("utf-8")).hexdigest()[:12]
```

```python
from __future__ import annotations
import datetime as dt
import json
import re
from pathlib import Path
from kb.dedup import fingerprint, fingerprint_source, normalize
from kb.entry import Entry, parse_entry, dump_entry
from kb.new_entry import create_entry

_MIN_FP_LEN = 16


class GateViolation(Exception):
    """门控拒绝: 试图做 Agent 不允许的写操作."""


class DuplicateCandidate(Exception):
    def __init__(self, entries: list[dict]):
        self.entries = entries
        super().__init__("duplicate candidates: "
                         + ", ".join(str(e.get("id")) for e in entries))


def _title_tokens(title: str) -> list[str]:
    toks = [t for t in re.split(r"[^0-9a-z\u4e00-\u9fff]+", title.lower()) if len(t) >= 2]
    return toks[:3]


def _index_records(repo: Path) -> list[dict]:
    jl = repo / "kb" / "index.jsonl"
    if not jl.exists():
        return []
    return [json.loads(x) for x in jl.read_text(encoding="utf-8").splitlines() if x.strip()]


def _tmp_entry(body: str) -> Entry:
    return Entry(fm={}, body=body, path=Path("<pending>"))


def _dup_hits(repos: list[Path], fp: str, title: str) -> list[dict]:
    hits: list[dict] = []
    seen: set[str] = set()
    toks = _title_tokens(title)
    for rp in repos:
        for r in _index_records(rp):
            rid = r.get("id")
            if rid in seen:
                continue
            hit = bool(fp) and r.get("dedup_key") == fp
            if not hit:
                hay = (r.get("title") or "").lower()
                hit = any(t in hay for t in toks)
            if hit:
                hits.append({**r, "layer": rp.name})
                seen.add(rid)
    return hits


def create_draft(repo, *, type: str, domain: str, title: str, body: str | None = None,
                 tags: list[str] | None = None, severity: str | None = None,
                 agent_kbs: tuple = (), today: dt.date | None = None) -> Entry:
    """Agent 唯一写入口: 强制 draft/speculative/project + 查重拒建."""
    today = today or dt.date.today()
    repos = [Path(repo)] + [Path(p) for p in agent_kbs]
    fp = ""
    if body:
        src = normalize(fingerprint_source(_tmp_entry(body)))
        fp = fingerprint(_tmp_entry(body)) if len(src) >= _MIN_FP_LEN else ""
    hits = _dup_hits(repos, fp, title)
    if hits:
        raise DuplicateCandidate(hits)
    e = create_entry(Path(repo), type=type, domain=domain, scope="project",
                     title=title, generated=True, today=today)
    if body or tags or severity:
        cur = parse_entry(e.path)
        if body:
            cur.body = f"\n# {title}\n\n{body.strip()}\n"
        if tags:
            cur.fm["tags"] = list(tags)
        if severity:
            cur.fm["severity"] = severity
        e.path.write_text(dump_entry(cur), encoding="utf-8")
        e = parse_entry(e.path)
    return e
```

```python
_TRANSITIONS: dict[str, set[str]] = {
    # spec §5.3: draft/proposed 均可由人工终审直达 accepted(accepted 时拒 speculative)
    "draft": {"proposed", "accepted", "deprecated"},
    "proposed": {"draft", "accepted", "deprecated"},
    "accepted": {"deprecated", "superseded"},
    "deprecated": set(),
    "superseded": set(),
}


def _find(repo, id: str) -> Path:
    kb = Path(repo) / "kb"
    for p in sorted(kb.rglob("*.md")) if kb.exists() else []:
        if p.name in ("INDEX.md",):
            continue
        if parse_entry(p).fm.get("id") == id:
            return p
    raise FileNotFoundError(f"entry not found: {id}")


def _save(p: Path, fm: dict, body: str, today: dt.date) -> Entry:
    fm["updated"] = today.isoformat()
    p.write_text(dump_entry(Entry(fm=fm, body=body, path=p)), encoding="utf-8")
    return parse_entry(p)


def update_draft(repo, id: str, *, title: str | None = None, body: str | None = None,
                 tags: list[str] | None = None, severity: str | None = None,
                 today: dt.date | None = None) -> Entry:
    p = _find(repo, id)
    e = parse_entry(p)
    if e.fm.get("status") != "draft":
        raise GateViolation(f"{id}: only draft entries are editable, current status={e.fm.get('status')}")
    fm = dict(e.fm)
    cur_body = e.body
    h1 = e.h1() or fm.get("title", "")
    if title:
        fm["title"] = title
        cur_body = cur_body.replace(f"# {h1}", f"# {title}", 1)
    if body:
        cur_body = f"\n# {fm.get('title', h1)}\n\n{body.strip()}\n"
    if tags is not None:
        fm["tags"] = list(tags)
    if severity:
        fm["severity"] = severity
    return _save(p, fm, cur_body, today or dt.date.today())


def review(repo, id: str, to: str, today: dt.date | None = None) -> Entry:
    p = _find(repo, id)
    e = parse_entry(p)
    frm = e.fm.get("status", "")
    if to not in _TRANSITIONS.get(frm, set()):
        raise GateViolation(f"{id}: illegal transition {frm} -> {to}")
    if to == "accepted" and e.fm.get("confidence") == "speculative":
        raise GateViolation(f"{id}: confidence=speculative must be verified before accepted")
    e.fm["status"] = to
    return _save(p, dict(e.fm), e.body, today or dt.date.today())


def set_scope(repo, id: str, scope: str, today: dt.date | None = None) -> Entry:
    if scope not in load_taxonomy()["enums"]["scope"]:
        raise ValueError(f"unknown scope: {scope}")
    p = _find(repo, id)
    e = parse_entry(p)
    fm = dict(e.fm)
    fm["scope"] = scope
    fm["tags"] = [t for t in (fm.get("tags") or []) if t != "needs-scope-review"]
    return _save(p, fm, e.body, today or dt.date.today())


def export_entry(project_repo, agent_kb_repo, id: str, *, to_scope: str,
                 today: dt.date | None = None) -> Path:
    if to_scope not in ("stack", "universal"):
        raise ValueError(f"export target scope must be stack/universal, got {to_scope}")
    src = _find(project_repo, id)
    rel = src.relative_to(Path(project_repo) / "kb")
    dst = Path(agent_kb_repo) / "kb" / rel
    if dst.exists():
        raise GateViolation(f"target already exists: {dst}")
    e = parse_entry(src)
    fm = dict(e.fm)
    fm["scope"] = to_scope
    fm["tags"] = [t for t in (fm.get("tags") or []) if t != "needs-scope-review"]
    fm["source"] = {**(fm.get("source") or {}),
                    "project": fm.get("source", {}).get("project") or Path(project_repo).resolve().name}
    dst.parent.mkdir(parents=True, exist_ok=True)
    dst.write_text(dump_entry(Entry(fm=fm, body=e.body, path=dst)), encoding="utf-8")
    src.unlink()
    return dst
```

```python
from __future__ import annotations
from pathlib import Path
from kb.query import get_entry, related
from kb.search import search


def _name(repo: Path) -> str:
    return Path(repo).resolve().name


def search_layers(repos: list[Path], *, type=None, domain=None, tag=None, scope=None,
                  status=None, confidence=None, query=None, limit: int = 10) -> list[dict]:
    """双层合并检索: repos 顺序即优先级, 逐层取满即截断(spec §7)."""
    out: list[dict] = []
    for rp in repos:
        if len(out) >= limit:
            break
        rows = search(rp, type=type, domain=domain, tag=tag, scope=scope,
                      status=status, confidence=confidence, query=query,
                      limit=limit - len(out))
        out.extend(rows)
    return out[:limit]


def get_layered(repos: list[Path], id: str) -> tuple[str, str] | None:
    for rp in repos:
        text = get_entry(rp, id)
        if text:
            return text, _name(rp)
    return None


def related_layers(repos: list[Path], id: str) -> list[dict]:
    out, seen = [], set()
    for rp in repos:
        for r in related(rp, id):
            if r["id"] not in seen:
                out.append(r)
                seen.add(r["id"])
    return out
```

                         ensure_ascii=False, indent=2), file=sys.stderr)
        return 1
    except (GateViolation, FileNotFoundError, ValueError) as exc:
        print(f"kb error: {exc}", file=sys.stderr)
        return 1

```python
from __future__ import annotations
import json
from pathlib import Path
from mcp.server.fastmcp import FastMCP
from kb.gate import DuplicateCandidate, GateViolation, create_draft, update_draft
from kb.layers import get_layered, related_layers, search_layers
from kb.query import get_entry


def _json(obj) -> str:
    return json.dumps(obj, ensure_ascii=False, indent=2)


def build_server(repo: Path, agent_kbs: list[Path] | None = None,
                 repo_name: str | None = None) -> FastMCP:
    repo = Path(repo)
    kbs = [Path(p) for p in (agent_kbs or [])]
    layers = [repo] + kbs
    name = repo_name or repo.resolve().name
    mcp = FastMCP(f"kb-{name}")

    @mcp.tool()
    def kb_search(query: str | None = None, type: str | None = None,
                  domain: str | None = None, tag: str | None = None,
                  scope: str | None = None, limit: int = 10) -> str:
        """检索知识库(项目层优先, 再通用层)。只返回元数据+摘要+路径, 不返回正文。
        draft 条目永不返回。命中后再用 kb_get 读全文。"""
        rows = search_layers(layers, type=type, domain=domain, tag=tag,
                             scope=scope, query=query, limit=limit)
        keep = ("id", "title", "type", "domain", "tags", "scope", "status",
                "severity", "confidence", "path", "summary", "updated", "repo")
        return _json([{k: r.get(k) for k in keep} for r in rows])

    @mcp.tool()
    def kb_get(id: str) -> str:
        """按 id 读取单条全文 markdown。"""
        got = get_layered(layers, id)
        if got is None:
            raise ValueError(f"not found: {id}")
        return got[0]

    @mcp.tool()
    def kb_related(id: str) -> str:
        """读取某条目的关联/被引用条目(双层, 只回元数据+摘要)。"""
        return _json(related_layers(layers, id))

    @mcp.tool()
    def kb_create(type: str, domain: str, title: str, body: str,
                  tags: list[str] | None = None, severity: str | None = None) -> str:
        """沉淀新知识。强制 status=draft / confidence=speculative / scope=project。
        查重命中会失败并返回候选 id, 此时应改用 kb_update。单次仅允许 1 条。"""
        try:
            e = create_draft(repo, type=type, domain=domain, title=title,
                             body=body, tags=tags, severity=severity,
                             agent_kbs=tuple(kbs))
        except DuplicateCandidate as exc:
            raise ValueError("duplicate candidates: "
                             + _json([{"id": h.get("id"), "title": h.get("title")}
                                      for h in exc.entries])
                             + " — use kb_update instead") from None
        return _json({**e.fm, "path": str(e.path)})

    @mcp.tool()
    def kb_update(id: str, title: str | None = None, body: str | None = None,
                  tags: list[str] | None = None, severity: str | None = None) -> str:
        """更新**本项目 draft 条目**。accepted/proposed 一律拒绝(人工 CLI 专用)。"""
        try:
            e = update_draft(repo, id, title=title, body=body, tags=tags, severity=severity)
        except FileNotFoundError:
            raise ValueError(f"not found in project layer: {id}") from None
        return _json({**e.fm, "path": str(e.path)})

    return mcp
```
